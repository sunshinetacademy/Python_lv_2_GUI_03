المستوى الأول 🟢 pack() — الرفوف 📦

أبسط مدير ترتيب. بيحط العناصر ورا بعض في اتجاه تختاره، وبيدي لكل عنصر «طرد» (parcel) خاص بيه.

pack_demo.py
import tkinter as tk

root = tk.Tk()
root.title("pack — الرفوف")
root.geometry("480x300")

# شريط أدوات: أزرار جنب بعض من الشمال
toolbar = tk.Frame(root, bg="#0b1f33")
toolbar.pack(side="top", fill="x")

for name in ["جديد", "فتح", "حفظ"]:
    tk.Button(toolbar, text=name).pack(
        side="left",      # جنب بعض
        padx=6, pady=6,   # مسافة خارجية
        ipadx=10, ipady=4 # حشو داخلي
    )

# محتوى يملأ باقي المساحة
body = tk.Label(root, text="المحتوى هنا", bg="#eef4fa")
body.pack(side="top", fill="both", expand=True)

# شريط حالة تحت
status = tk.Label(root, text="جاهز ✅", bg="#dbeafe")
status.pack(side="bottom", fill="x")

root.mainloop()
الباراميترالقيمبيعمل إيه؟
sidetop / bottom / left / rightالناحية اللي العنصر بيلزق فيها
fillnone / x / y / bothيمتد أفقياً / رأسياً / الاتنين
expandTrue / Falseياخد نصيبه من المساحة الفاضية
padx / padyرقم بالبكسلمسافة خارج العنصر
ipadx / ipadyرقم بالبكسلحشو داخل العنصر (بيكبّره)
💡 الفرق الذهبي: pady = مسافة بين العنصر وجاره (برّه). ipady = العنصر نفسه يبقى أطول (جوّه). جرّب الاتنين في المختبر وشوف الفرق! 🔍
⚠️ فخ: fill="both" لوحدها مش كفاية! لازم expand=True كمان عشان العنصر ياخد المساحة الفاضية ويملاها.

المستوى الثاني 🟡 grid() — الجدول 🔲

أقوى مدير ترتيب للواجهات الحقيقية. بيقسّم الحاوية لـ صفوف وأعمدة زي ورقة المربعات، وكل عنصر بياخد خلية أو أكتر.

grid_demo.py
import tkinter as tk

root = tk.Tk()
root.title("grid — الجدول")
root.geometry("420x220")

# صف 0: عنوان ياخد العمودين
tk.Label(root, text="بيانات الطالب",
         font=("Arial", 14, "bold")).grid(
    row=0, column=0, columnspan=2, pady=12, sticky="w")

# صف 1 و 2: لیبل + حقل إدخال
tk.Label(root, text="الاسم").grid(row=1, column=0, sticky="w", padx=12, pady=6)
name = tk.Entry(root)
name.grid(row=1, column=1, sticky="ew", padx=12, pady=6, ipady=4)

tk.Label(root, text="العمر").grid(row=2, column=0, sticky="w", padx=12, pady=6)
age = tk.Entry(root)
age.grid(row=2, column=1, sticky="ew", padx=12, pady=6, ipady=4)

# 🔑 السر: العمود التاني يكبر مع النافذة
root.grid_columnconfigure(1, weight=1)
root.grid_rowconfigure(3, weight=1)

root.mainloop()

🧭 sticky — بوصلة الالتصاق

sticky بتقول للعنصر يلزق في أنهي ناحية من الخلية بتاعته. الحروف اختصارات اتجاهات البوصلة:

القيمةالمعنىالنتيجة
"w"West = غرب = شماليلزق على الشمال
"e"East = شرق = يمينيلزق على اليمين
"n" / "s"North / Southفوق / تحت
"ew"شرق + غربيمتد أفقياً بعرض الخلية
"ns"شمال + جنوبيمتد رأسياً
"nsew"الأربع اتجاهاتيملا الخلية بالكامل ⭐
🌟 weight — سر الواجهة المتجاوبة: افتراضياً الأعمدة مش بتكبر لما تكبّر النافذة! لازم تقول root.grid_columnconfigure(i, weight=1) عشان العمود ياخد نصيبه من المساحة الجديدة. من غير weight الواجهة بتفضل صغيرة في النص. 📏
📐 columnspan / rowspan: لو عايز عنصر ياخد عرض عمودين → columnspan=2. ولو عايز ياخد ارتفاع صفين → rowspan=2. زي دمج الخلايا في Excel بالظبط! 🧮

المستوى الثالث 🔴 place() — المسمار 📍

بتحدد مكان العنصر بإحداثيات صريحة. دقيقة جداً… بس خطيرة لأنها مش بتتأقلم مع تكبير النافذة.

place_demo.py
import tkinter as tk

root = tk.Tk()
root.title("place — المسمار")
root.geometry("460x300")
root.configure(bg="#eef4fa")

# 1) إحداثيات مطلقة بالبكسل
tk.Label(root, text="شعار", bg="#fbbf24").place(
    x=20, y=20, width=90, height=32, anchor="nw")

# 2) إحداثيات نسبية (0.0 إلى 1.0) — تتأقلم شوية
tk.Button(root, text="في النص تماماً").place(
    relx=0.5, rely=0.5, anchor="center",
    width=160, height=40)

# 3) في الزاوية تحت-يمين
tk.Label(root, text="v1.0", bg="#dbeafe").place(
    relx=1.0, rely=1.0, anchor="se", x=-10, y=-10)

root.mainloop()
الباراميتربيعمل إيه؟
x, yموقع بالبكسل من أعلى-شمال الحاوية
relx, relyموقع نسبي: 0.0 = البداية، 0.5 = النص، 1.0 = النهاية
anchorأنهي نقطة من العنصر تتثبت: nw / n / ne / w / center / e / sw / s / se
width, heightحجم صريح بالبكسل
relwidth, relheightحجم نسبي من الحاوية
⚠️ امتى نتجنب place؟ لو المستخدم كبّر النافذة، العناصر مش بتتحرك وممكن تتقص أو تتراكب. استخدمها بس لحاجة ثابتة زي شعار أو علامة إصدار — مش للواجهة كلها! 🚫
✅ الاستخدام الصح: place جوه Frame صغير مخصص، والباقي كله grid أو pack. كده تستفيد من الدقة من غير ما تبوظ الواجهة.

المستوى الرابع 🟣 Frame — القواطع 🖼️

🚨 القاعدة الذهبية للمحاضرة: ممنوع تستخدم pack() و grid() مع بعض جوّه نفس الأب! Tkinter بيدخل في صراع حسابات والبرنامج يتجمّد (hang) — مش رسالة خطأ، تجميد كامل! 😱

الحل؟ الـ Frame! حاوية بتقسم النافذة لأقسام، وكل قسم يقدر يستخدم مدير الترتيب اللي هو عايزه — لأن كل Frame أب مستقل.

frames_demo.py
import tkinter as tk

root = tk.Tk()
root.title("Frame — القواطع")
root.geometry("560x340")

# الأب root بيستخدم pack فقط
header = tk.Frame(root, bg="#2563eb", height=54)
header.pack(side="top", fill="x")
header.pack_propagate(False)   # نحافظ على الارتفاع

tk.Label(header, text="☀️ Sunshine T Academy",
         bg="#2563eb", fg="white",
         font=("Arial", 14, "bold")).pack(side="right", padx=16)

# قسم وسط: هنا بنستخدم grid بأمان (أب مختلف!)
middle = tk.Frame(root, bg="#eef4fa")
middle.pack(side="top", fill="both", expand=True)

for i in range(6):
    tk.Button(middle, text=f"خلية {i+1}").grid(
        row=i // 3, column=i % 3,
        sticky="nsew", padx=8, pady=8, ipady=8)

for c in range(3):
    middle.grid_columnconfigure(c, weight=1)
for r in range(2):
    middle.grid_rowconfigure(r, weight=1)

# فوتر: pack تاني على root — تمام لأنه نفس المدير
footer = tk.Frame(root, bg="#0b1f33", height=32)
footer.pack(side="bottom", fill="x")
footer.pack_propagate(False)

root.mainloop()
🧠 запомни: المدير الواحد لكل أب. root كلها pack ✅ — middle كلها grid ✅ — مفيش تعارض لأنهم آباء مختلفين! 🎯

# ═══════════════════════════════════════
# 🔐 Login Form — Sunshine T Academy
# المحاضرة 8: Layout Managers
# ═══════════════════════════════════════
import tkinter as tk
from tkinter import messagebox

BG   = "#eef4fa"
CARD = "#ffffff"
INK  = "#0b1f33"
BLUE = "#2563eb"


def try_login():
    user = user_entry.get().strip()
    pwd  = pwd_entry.get()

    if not user or not pwd:
        messagebox.showwarning("ناقص بيانات", "اكتب المستخدم وكلمة السر")
        return

    if user == "admin" and pwd == "1234":
        messagebox.showinfo("نجاح", f"أهلاً {user}! ✅")
    else:
        messagebox.showerror("فشل", "بيانات غير صحيحة ❌")


def clear_fields():
    user_entry.delete(0, tk.END)
    pwd_entry.delete(0, tk.END)
    user_entry.focus()


# ===== النافذة =====
root = tk.Tk()
root.title("تسجيل الدخول — Sunshine T Academy")
root.geometry("470x370")
root.minsize(410, 330)
root.configure(bg=BG)

# root كلها grid — صف وعمود واحد بيكبر
root.grid_rowconfigure(0, weight=1)
root.grid_columnconfigure(0, weight=1)

# ===== الكارت =====
card = tk.Frame(root, bg=CARD, padx=24, pady=20)
card.grid(row=0, column=0, padx=24, pady=24, sticky="nsew")

card.grid_columnconfigure(0, weight=0)   # الليبلز ثابتة
card.grid_columnconfigure(1, weight=1)   # الحقول تتمدد ⭐

title = tk.Label(card, text="☀️ تسجيل الدخول",
                 font=("Arial", 18, "bold"), bg=CARD, fg=BLUE)
title.grid(row=0, column=0, columnspan=2, sticky="w", pady=(0, 4))

sub = tk.Label(card, text="Layout Managers — grid + Frame",
               font=("Arial", 10), bg=CARD, fg="#94a3b8")
sub.grid(row=1, column=0, columnspan=2, sticky="w", pady=(0, 18))

# ===== الصفوف =====
tk.Label(card, text="المستخدم", bg=CARD, fg=INK,
         font=("Arial", 11, "bold")).grid(
    row=2, column=0, sticky="w", pady=8, padx=(0, 10))

user_entry = tk.Entry(card, font=("Arial", 12))
user_entry.grid(row=2, column=1, sticky="ew", pady=8, ipady=5)

tk.Label(card, text="كلمة السر", bg=CARD, fg=INK,
         font=("Arial", 11, "bold")).grid(
    row=3, column=0, sticky="w", pady=8, padx=(0, 10))

pwd_entry = tk.Entry(card, font=("Arial", 12), show="•")
pwd_entry.grid(row=3, column=1, sticky="ew", pady=8, ipady=5)

# ===== Frame للأزرار: هنا pack بأمان (أب مختلف) =====
buttons = tk.Frame(card, bg=CARD)
buttons.grid(row=4, column=0, columnspan=2, sticky="ew", pady=(20, 0))

login_btn = tk.Button(buttons, text="دخول", bg=BLUE, fg="white",
                      font=("Arial", 12, "bold"), relief="flat",
                      command=try_login)
login_btn.pack(side="right", fill="x", expand=True,
               padx=(6, 0), ipady=7)

clear_btn = tk.Button(buttons, text="مسح", bg="#e2e8f0", fg=INK,
                      font=("Arial", 12), relief="flat",
                      command=clear_fields)
clear_btn.pack(side="right", fill="x", expand=True,
               padx=(0, 6), ipady=7)

hint = tk.Label(card, text="جرّب: admin / 1234",
                bg=CARD, fg="#94a3b8", font=("Arial", 9))
hint.grid(row=5, column=0, columnspan=2, sticky="w", pady=(14, 0))

# الصف الفاضي ياخد باقي المساحة
card.grid_rowconfigure(6, weight=1)

user_entry.focus()
root.bind("<Return>", lambda e: try_login())

root.mainloop()

🎯 5 مهام مميزة — تحدي المبرمجين!

    

      
🥉 المهمة 1: شريط أدوات 20 نقطة

      
اعمل نافذة فيها 4 أزرار جنب بعض من الشمال باستخدام pack(side="left") مع padx=6 و ipadx=12.

      
💡 تلميح: حط الأزرار جوّه Frame وخليه pack(fill="x")

    

    

      
🥉 المهمة 2: لوحة 2×3 25 نقطة

      
اعمل 6 أزرار في صفين و3 أعمدة بـ grid، وكل زر يملا خليته بالكامل ويكبر مع النافذة.

      
💡 تلميح: sticky="nsew" + grid_columnconfigure(c, weight=1) لكل عمود + weight للصفوف

    

    

      
🥈 المهمة 3: نموذج بيانات 35 نقطة

      
اعمل نموذج فيه: عنوان ياخد عمودين، و3 صفوف (الاسم / العمر / المدينة) كل صف ليبل + Entry. الحقول تتمدد لما تكبّر النافذة.

      
💡 تلميح: columnspan=2 للعنوان + sticky="ew" للحقول + weight=1 للعمود التاني

    

    

      
🥈 المهمة 4: واجهة 3 أقسام 40 نقطة

      
اقسم النافذة لـ: هيدر ثابت فوق، محتوى في النص (شبكة أزرار)، فوتر ثابت تحت. استخدم pack على النافذة و grid جوّه المحتوى.

      
💡 تلميح: 3 × Frame + pack_propagate(False) للهيدر والفوتر عشان يحافظوا على ارتفاعهم

    

    

      
🎨 المهمة 5: لوحة تحكم كاملة 60 نقطة

      
صمّم لوحة تحكم فيها: هيدر فيه شعار بـ pack، سايدبار على الشمال فيه 4 أزرار، منطقة محتوى بشبكة 3×2، فوتر فيه حالة البرنامج، وشعار صغير في زاوية المحتوى بـ place. كل حاجة تتجاوب مع التكبير.

      
💡 تلميح: root pack → middle Frame فيه pack (سايدبار + content) → content Frame فيه grid → الشعار place(relx=1, rely=1, anchor="se")

    

    
📦 طريقة التسليم: ملف .py + صورة للنافذة وهي صغيرة وأخرى وهي مكبّرة (عشان ن证明 الـ weight شغال) + سطرين تشرح فيهم ليه استخدمت Frame. ارفعهم قبل المحاضرة الجاية! 📤
